Step-by-step
Publishing to GitHub Pages + Google Apps Script
Uses your existing NVIDIA Google account — no new services or billing needed
1
Create a Google Sheet
Go to sheets.new in your NVIDIA Google account to create a new spreadsheet. Name it something like Ali Get Well Wishes. This sheet will be the database that stores all team messages.
2
Open Apps Script
In the Google Sheet → click Extensions in the top menu → click Apps Script. A new tab opens with a code editor. Delete any existing code in the editor.
3
Paste the Script Code
Copy the entire block below and paste it into the Apps Script editor, then click the 💾 Save icon (or Ctrl+S): 
const SHEET_NAME = 'Wishes';

function getSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) {
    sh = ss.insertSheet(SHEET_NAME);
    sh.appendRow(['id','name','role','text','ts']);
    sh.setFrozenRows(1);
  }
  return sh;
}

function doGet(e) {
  const action = e.parameter.action || 'get';
  const sh = getSheet();

  if (action === 'get') {
    const vals = sh.getDataRange().getValues();
    if (vals.length < 2) return _json([]);
    const hdrs = vals[0];
    return _json(vals.slice(1).map(r =>
      Object.fromEntries(hdrs.map((h, i) => [h, r[i]]))));
  }

  if (action === 'delete') {
    const id = e.parameter.id;
    const vals = sh.getDataRange().getValues();
    for (let i = 1; i < vals.length; i++) {
      if (String(vals[i][0]) === String(id)) {
        sh.deleteRow(i + 1);
        return _json({ ok: true });
      }
    }
    return _json({ error: 'not found' });
  }

  return _json({ error: 'unknown action' });
}

function doPost(e) {
  try {
    const b = JSON.parse(e.postData.contents);
    const sh = getSheet();
    const id = Date.now().toString();
    sh.appendRow([id, b.name || '', b.role || '', b.text || '',
                  new Date().toISOString()]);
    return _json({ ok: true, id });
  } catch (err) {
    return _json({ error: String(err) });
  }
}

function _json(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
4
Deploy as a Web App
In the Apps Script editor → click the blue Deploy button (top right) → New deployment → click the gear ⚙️ next to "Select type" and choose Web app.

Fill in the deployment settings exactly as follows:
  • Description: Ali Wishes API
  • Execute as: Me (your NVIDIA account)
  • Who has access: Anyone
Click Deploy. You may be asked to grant permissions — click Authorize access and follow the prompts.
5
Copy the Web App URL
After deploying, a dialog shows Web app URL — it looks like:
https://script.google.com/macros/s/AKfycb.../exec
Copy this URL — you'll paste it into the HTML file next.
6
Paste the URL into the HTML File
Open Get Well Soon - Ali.html in any text editor → find the SCRIPT_URL line near the top of the script section → replace the placeholder: 
// Before:
const SCRIPT_URL = 'https://script.google.com/a/macros/nvidia.com/s/AKfycbzpazjdzic9dQhRO5_F-K10lDCZ8VBZtukCjQUQk-jDhPAIHLF1qy2bHV-TIjXDr7Yfew/exec';

// After:
const SCRIPT_URL = 'https://script.google.com/macros/s/AKfycb.../exec';
Save the file. Wishes will now be stored in your Google Sheet and visible to everyone who opens the card.
7
Publish to GitHub Pages (optional)
To share via a link: go to github.com/new → create a Public repository → upload this HTML file → go to repository Settings → Pages → set source to main branch / (root) → Save.
Your shareable link will be: https://YOUR-USERNAME.github.io/REPO-NAME/Get Well Soon - Ali.html

All done! Every teammate who opens the link will see all existing wishes live, and new ones appear automatically every 15 seconds. 🎉
